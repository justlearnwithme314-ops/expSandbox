cmake_minimum_required(VERSION 3.24)
project(VoxelLab VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(VOXEL_BUILD_CLIENT "Build the graphical game" ON)
option(VOXEL_BUILD_TESTS "Build world and mesher tests" ON)
add_library(voxel_sqlite STATIC references/SQLiteCpp/sqlite3/sqlite3.c)
target_include_directories(voxel_sqlite PUBLIC references/SQLiteCpp/sqlite3)
if(UNIX)
  target_link_libraries(voxel_sqlite PUBLIC ${CMAKE_DL_LIBS} pthread m)
endif()
add_library(voxel_core STATIC src/world.cpp src/mesher.cpp src/gameplay.cpp)
set_target_properties(voxel_core voxel_sqlite PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(voxel_core PUBLIC src references/json/single_include PRIVATE references/FastNoiseLite/Cpp)
target_link_libraries(voxel_core PUBLIC voxel_sqlite)
add_library(voxel-api SHARED src/c_api.cpp)
target_link_libraries(voxel-api PRIVATE voxel_core)
target_compile_definitions(voxel-api PRIVATE VOXEL_API_BUILD)
add_executable(voxel-cli src/cli.cpp)
target_link_libraries(voxel-cli PRIVATE voxel_core)
if(MSVC)
  target_compile_options(voxel_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(voxel_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(VOXEL_BUILD_CLIENT)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
  add_subdirectory(references/raylib EXCLUDE_FROM_ALL)
  add_executable(voxel-lab src/main.cpp)
  target_link_libraries(voxel-lab PRIVATE voxel_core raylib)
  target_compile_definitions(voxel-lab PRIVATE VOXEL_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  set_target_properties(voxel-lab PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
if(VOXEL_BUILD_TESTS)
  enable_testing()
  add_executable(voxel-tests tests/core_tests.cpp)
  target_link_libraries(voxel-tests PRIVATE voxel_core)
  target_compile_definitions(voxel-tests PRIVATE VOXEL_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME world_gameplay_transvoxel COMMAND voxel-tests)
endif()
